How to Use:

const response = await fetch('/api/email', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    to: 'newuser@example.com',
    subject: 'Welcome to Kalvium!',
    message: welcomeTemplate('John Doe')
  })
});
